접힌 선으로 2 · 3 페이지 자르기

접힌 선이 책 맨 위와 맨 아래에 닿는 두 점 S, E 를 중학교 일차식으로 구하고, 그 두 점으로 clip-path 다각형을 만들어 2 · 3 페이지를 자릅니다. "다음" 을 누르며 보세요.

    <!-- 책 상자. 좌표는 전부 책 왼쪽 위 (0, 0) 기준 -->
<div class="book">                 <!-- 1000 × 700 -->
  <div class="page page1"></div>
  <div class="page page4"></div>
  <div class="pages">              <!-- 책 크기 부모. 돌리지 않음. clip-path 는 여기 -->
    <div class="page page2"></div>
    <div class="page page3"></div>  <!-- 첫 번째 문서대로 A′ 에 두고 돌림 -->
  </div>
</div>
    // P, F 는 { x, y } 좌표. H 는 책 세로 700
const M = { x: (P.x + F.x) / 2, y: (P.y + F.y) / 2 };   // (880, 290)

const pfSlope = (F.y - P.y) / (F.x - P.x);   // P–F 기울기      −0.75
const a = -1 / pfSlope;                      // 접힌 선 기울기   1.333
const b = M.y - a * M.x;                     // M 을 지나게     −883.3   →  접힌 선 y = a x + b

const S = { x: (0 - b) / a, y: 0 };          // 맨 위 줄 y = 0     (662.5, 0)
const E = { x: (H - b) / a, y: H };          // 맨 아래 줄 y = 700  (1187.5, 700)

// 수평으로 끌면(F.y === P.y) 접힌 선이 세로선이라 기울기를 못 구한다. 이때는 S.x = E.x = M.x
if (F.y === P.y) {
  S.x = M.x;
  E.x = M.x;
}

// 2 · 3 페이지를 담은 부모에 건다: S → E → 책 왼쪽 아래 → 책 왼쪽 위
pages.style.clipPath = `polygon(${S.x}px 0px, ${E.x}px ${H}px, 0px ${H}px, 0px 0px)`;
